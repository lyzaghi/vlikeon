cmake_minimum_required(VERSION 3.25)
project(OmniPlayMemoryFilesQualification LANGUAGES CXX)
# Configure/build/execute only in item 16 after all product source is authored.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(OMNIPLAY_NATIVE_MEMORY_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../engines/native/src" CACHE PATH "Exact current native frontend source directory")
set(native "${OMNIPLAY_NATIVE_MEMORY_SOURCE}")
if(NOT EXISTS "${native}/memory_files.cpp" OR NOT EXISTS "${native}/writable_files_v1.h")
  message(FATAL_ERROR "Choose the exact authored native frontend source directory")
endif()
add_executable(omniplay-memory-files-cases memory_files_cases.cpp "${native}/memory_files.cpp")
target_include_directories(omniplay-memory-files-cases PRIVATE "${native}")
if(MSVC)
  target_compile_definitions(omniplay-memory-files-cases PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(omniplay-memory-files-cases PRIVATE /W4 /permissive- /EHsc /utf-8 /GS /guard:cf)
  target_link_options(omniplay-memory-files-cases PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
  target_compile_options(omniplay-memory-files-cases PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
endif()
enable_testing()
add_test(NAME memory-files-contract COMMAND omniplay-memory-files-cases)
