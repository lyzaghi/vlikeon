cmake_minimum_required(VERSION 3.25)
project(OmniPlayWindowsPrivatePeerQualification LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This actual contained-process qualification targets Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OMNIPLAY_PRESENTATION_SOURCE "" CACHE PATH "Exact reviewed presentation source")
if(NOT EXISTS "${OMNIPLAY_PRESENTATION_SOURCE}/src/wire.cpp")
  message(FATAL_ERROR "Explicit production wire source is required")
endif()
add_executable(omniplay-presentation-fixture presentation-fixture.cpp "${OMNIPLAY_PRESENTATION_SOURCE}/src/wire.cpp")
target_include_directories(omniplay-presentation-fixture PRIVATE "${OMNIPLAY_PRESENTATION_SOURCE}/src")
target_compile_definitions(omniplay-presentation-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(omniplay-presentation-fixture PRIVATE /W4 /permissive- /EHsc /utf-8 /GS /guard:cf)
  target_link_options(omniplay-presentation-fixture PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
endif()
