cmake_minimum_required(VERSION 3.25)
project(OmniPlayNativeAudioQualification LANGUAGES CXX)
# Source authoring only. Configure/build/execute in the final item-16 pass.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(OMNIPLAY_NATIVE_AUDIO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../engines/native/src" CACHE PATH "Exact authored native frontend headers")
if(NOT EXISTS "${OMNIPLAY_NATIVE_AUDIO_SOURCE}/audio_ring.hpp")
  message(FATAL_ERROR "Choose the exact native frontend source directory")
endif()
find_package(Threads REQUIRED)
add_executable(omniplay-native-audio-cases audio_ring_cases.cpp)
target_include_directories(omniplay-native-audio-cases PRIVATE "${OMNIPLAY_NATIVE_AUDIO_SOURCE}")
target_link_libraries(omniplay-native-audio-cases PRIVATE Threads::Threads)
if(MSVC)
  target_compile_definitions(omniplay-native-audio-cases PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(omniplay-native-audio-cases PRIVATE /W4 /permissive- /EHsc /utf-8 /GS /guard:cf)
  target_link_options(omniplay-native-audio-cases PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
  target_compile_options(omniplay-native-audio-cases PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
endif()
enable_testing()
add_test(NAME native-audio-ring-contract COMMAND omniplay-native-audio-cases)
