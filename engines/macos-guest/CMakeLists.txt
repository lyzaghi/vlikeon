cmake_minimum_required(VERSION 3.25)
project(VlikeonGuestInit LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Guest init requires an explicit Linux ARM64 toolchain")
endif()
add_executable(vlikeon-guest-init init.c)
set_target_properties(vlikeon-guest-init PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(vlikeon-guest-init PRIVATE _GNU_SOURCE)
target_compile_options(vlikeon-guest-init PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
target_link_options(vlikeon-guest-init PRIVATE -static -Wl,-z,relro,-z,now,-z,noexecstack)
