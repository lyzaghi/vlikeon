cmake_minimum_required(VERSION 3.25)
project(OmniPlayLinuxIsolationPolicy LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "This policy source requires an explicitly qualified Linux x86-64 toolchain")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
# Preserve an exact upstream UAPI even on distributions with older SDK headers.
# This changes compilation inputs, never the runtime ABI >= 6 admission.
function(require_uapi_file relative expected)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/uapi/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned Linux UAPI source/license mismatch: ${relative}")
  endif()
endfunction()
require_uapi_file("linux/landlock.h" "ea5575c369326da8fef2e62d0a9b528b45f4aa2302f836e995cb6d5c33bacc49")
require_uapi_file("licenses/GPL-2.0.txt" "f6b78c087c3ebdf0f3c13415070dd480a3f35d8fc76f3d02180a407c1c812f79")
require_uapi_file("licenses/Linux-syscall-note.txt" "8e378ab93586eb55135d3bc119cce787f7324f48394777d00c34fa3d0be3303f")
add_library(omniplay-linux-policy STATIC policy.c)
target_include_directories(omniplay-linux-policy PUBLIC .)
target_compile_definitions(omniplay-linux-policy PRIVATE _GNU_SOURCE)
target_compile_options(omniplay-linux-policy PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong -fPIC)
add_library(omniplay-linux-namespaces STATIC namespaces.c root.c cgroup.c)
target_include_directories(omniplay-linux-namespaces PUBLIC .)
target_link_libraries(omniplay-linux-namespaces PUBLIC omniplay-linux-policy)
target_compile_definitions(omniplay-linux-namespaces PRIVATE _GNU_SOURCE)
target_compile_options(omniplay-linux-namespaces PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong -fPIC)
add_executable(omniplay-linux-bootstrap bootstrap.c)
target_link_libraries(omniplay-linux-bootstrap PRIVATE omniplay-linux-namespaces)
target_compile_definitions(omniplay-linux-bootstrap PRIVATE _GNU_SOURCE)
target_compile_options(omniplay-linux-bootstrap PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong -fPIE)
target_link_options(omniplay-linux-bootstrap PRIVATE -static-pie -Wl,-z,relro,-z,now,-z,noexecstack)
