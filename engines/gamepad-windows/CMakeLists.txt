cmake_minimum_required(VERSION 3.25)
project(VlikeonWinGamepad LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The WinUHid adapter requires 64-bit Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(WINUHID_SOURCE "" CACHE PATH "Explicit local checkout of the pinned WinUHid source")
if(NOT IS_DIRECTORY "${WINUHID_SOURCE}")
  message(FATAL_ERROR "Provide WINUHID_SOURCE; this recipe never downloads or installs a driver")
endif()
file(REAL_PATH "${WINUHID_SOURCE}" upstream)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/winuhid-source.json" provenance)
set(staged "${CMAKE_CURRENT_BINARY_DIR}/pinned-winuhid")
string(JSON count LENGTH "${provenance}" files)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON relative MEMBER "${provenance}" files ${index})
  string(JSON expected GET "${provenance}" files "${relative}")
  set(candidate "${upstream}/${relative}")
  if(NOT EXISTS "${candidate}" OR IS_DIRECTORY "${candidate}" OR IS_SYMLINK "${candidate}")
    message(FATAL_ERROR "Missing or linked upstream file: ${relative}")
  endif()
  file(REAL_PATH "${candidate}" resolved)
  cmake_path(IS_PREFIX upstream "${resolved}" NORMALIZE inside)
  if(NOT inside)
    message(FATAL_ERROR "Upstream file escaped the explicit checkout")
  endif()
  file(SHA256 "${candidate}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned upstream source mismatch: ${relative}")
  endif()
  # Compile a selected-file copy, not the checkout's whole include directory.
  # In particular an extra checkout windows.h must not shadow the Windows SDK.
  get_filename_component(parent "${staged}/${relative}" DIRECTORY)
  file(MAKE_DIRECTORY "${parent}")
  configure_file("${candidate}" "${staged}/${relative}" COPYONLY)
  file(SHA256 "${staged}/${relative}" copied)
  if(NOT copied STREQUAL expected)
    message(FATAL_ERROR "Copied source differs from the locked bytes: ${relative}")
  endif()
endforeach()
# Source must be an immutable, admitted staging input. Hashing is not protection
# against a concurrent writer and does not admit the separately installed driver.
add_library(vlikeon-winpad STATIC src/gamepad.cpp "${staged}/WinUHid/WinUHid.cpp")
target_include_directories(vlikeon-winpad PUBLIC include PRIVATE "${staged}/WinUHid")
target_compile_definitions(vlikeon-winpad PRIVATE WINUHID_STATIC _WIN32_WINNT=0x0A00)
target_compile_options(vlikeon-winpad PRIVATE /W4 /permissive- /EHsc /GS /guard:cf)
set_property(TARGET vlikeon-winpad PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
target_link_libraries(vlikeon-winpad PUBLIC setupapi hid ole32)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/notices")
configure_file("${staged}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/notices/WinUHid-LICENSE.txt" COPYONLY)
# This target is source authoring only. No install, driver-signature acceptance,
# device operation, product feature activation or release packaging is defined.
