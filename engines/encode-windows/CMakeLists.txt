cmake_minimum_required(VERSION 3.25)
project(VlikeonWindowsEncoder LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native encoder requires 64-bit Windows and MSVC")
endif()
add_library(vlikeon-windows-encoder STATIC src/encoder.cpp src/capture.cpp)
target_include_directories(vlikeon-windows-encoder PUBLIC include)
target_compile_features(vlikeon-windows-encoder PUBLIC cxx_std_20)
target_compile_definitions(vlikeon-windows-encoder PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(vlikeon-windows-encoder PRIVATE /W4 /EHsc /guard:cf /sdl /bigobj)
target_link_libraries(vlikeon-windows-encoder PUBLIC mfplat mf mfuuid ole32 oleaut32 strmiids
  d3d11 dxgi user32 dwmapi windowsapp runtimeobject)
add_executable(omniplay-video-worker src/worker.cpp)
target_link_libraries(omniplay-video-worker PRIVATE vlikeon-windows-encoder)
target_compile_options(omniplay-video-worker PRIVATE /W4 /EHsc /guard:cf /sdl)
target_link_options(omniplay-video-worker PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
