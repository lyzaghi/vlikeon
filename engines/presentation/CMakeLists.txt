cmake_minimum_required(VERSION 3.25)
project(OmniPlayPresentationHost LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
add_library(omniplay-presentation-wire STATIC src/wire.cpp)
target_include_directories(omniplay-presentation-wire PUBLIC src)
add_library(omniplay-presentation-client STATIC src/client.cpp)
target_link_libraries(omniplay-presentation-client PUBLIC omniplay-presentation-wire Threads::Threads)
target_include_directories(omniplay-presentation-client PUBLIC src)
target_compile_definitions(omniplay-presentation-wire PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set(OMNIPLAY_PRESENTATION_TARGETS omniplay-presentation-wire omniplay-presentation-client)
option(OMNIPLAY_BUILD_PRESENTATION_HOST "Build the trusted SDL presentation host" ON)
if(OMNIPLAY_BUILD_PRESENTATION_HOST)
  set(OMNIPLAY_SDL_SOURCE "" CACHE PATH "Explicit pinned SDL 3.4.18 source")
  if(NOT EXISTS "${OMNIPLAY_SDL_SOURCE}/include/SDL3/SDL.h")
    message(FATAL_ERROR "A complete explicit pinned SDL source is required")
  endif()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  if(NOT TARGET SDL3::SDL3-static)
    add_subdirectory("${OMNIPLAY_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
  endif()
  add_executable(omniplay-presentation-host src/main.cpp src/owner.cpp)
  target_link_libraries(omniplay-presentation-host PRIVATE omniplay-presentation-wire SDL3::SDL3-static Threads::Threads)
  target_compile_definitions(omniplay-presentation-host PRIVATE SDL_MAIN_HANDLED NOMINMAX WIN32_LEAN_AND_MEAN)
  list(APPEND OMNIPLAY_PRESENTATION_TARGETS omniplay-presentation-host)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND TARGET omniplay-presentation-host)
  set_target_properties(omniplay-presentation-host PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "")
  target_compile_definitions(omniplay-presentation-host PRIVATE _GNU_SOURCE)
  target_compile_options(omniplay-presentation-host PRIVATE -fPIE)
  target_link_options(omniplay-presentation-host PRIVATE -pie -Wl,-z,relro,-z,now,-z,noexecstack)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin" AND TARGET omniplay-presentation-host)
  target_link_options(omniplay-presentation-host PRIVATE -Wl,-pie -Wl,-dead_strip -Wl,-fatal_warnings)
  set_target_properties(omniplay-presentation-host PROPERTIES
    BUILD_RPATH "" INSTALL_RPATH "" XCODE_ATTRIBUTE_ENABLE_HARDENED_RUNTIME YES
    XCODE_ATTRIBUTE_CODE_SIGN_INJECT_BASE_ENTITLEMENTS NO
    XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/macos-presentation.entitlements")
endif()
foreach(target IN LISTS OMNIPLAY_PRESENTATION_TARGETS)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /GS /guard:cf /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  endif()
endforeach()
if(MSVC AND TARGET omniplay-presentation-host)
  target_sources(omniplay-presentation-host PRIVATE host.manifest)
  target_link_options(omniplay-presentation-host PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
endif()
