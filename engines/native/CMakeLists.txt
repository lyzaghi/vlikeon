cmake_minimum_required(VERSION 3.25)
project(OmniPlayNativeWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(OMNIPLAY_SDL_SOURCE "" CACHE PATH "Pinned SDL source checkout")
set(OMNIPLAY_JSON_SOURCE "" CACHE PATH "Pinned nlohmann/json source checkout")
set(OMNIPLAY_LIBRETRO_INCLUDE "" CACHE PATH "Reviewed MIT libretro interface directory")
set(OMNIPLAY_RUNTIME_ID "" CACHE STRING "Approved runtime catalog identity")
set(OMNIPLAY_CORE_FILENAME "" CACHE STRING "Core library basename beside the worker")
set(OMNIPLAY_VIDEO_BACKEND "software" CACHE STRING "Declared native presentation adapter")
option(OMNIPLAY_DEVICE_FREE_ACTOR "Software actor with private presentation and no SDL/device linkage" OFF)
option(OMNIPLAY_GUEST_ACTOR "Device-free Linux ARM64 actor for the diskless macOS guest" OFF)
if(OMNIPLAY_GUEST_ACTOR)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$" OR NOT OMNIPLAY_DEVICE_FREE_ACTOR)
    message(FATAL_ERROR "Guest actor requires an explicit Linux ARM64 device-free toolchain")
  endif()
endif()
option(OMNIPLAY_BUILD_PRESENTATION_HOST "Also build the reviewed private presentation helper" OFF)
if(OMNIPLAY_DEVICE_FREE_ACTOR AND ((NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT OMNIPLAY_VIDEO_BACKEND STREQUAL "software"))
  message(FATAL_ERROR "Device-free actor requires a supported Windows/macOS/Linux software target and private presentation")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin" AND NOT OMNIPLAY_DEVICE_FREE_ACTOR)
  message(FATAL_ERROR "macOS actors require the device-free private presentation route")
endif()
foreach(required OMNIPLAY_SDL_SOURCE OMNIPLAY_JSON_SOURCE OMNIPLAY_LIBRETRO_INCLUDE OMNIPLAY_RUNTIME_ID OMNIPLAY_CORE_FILENAME)
  if("${${required}}" STREQUAL "")
    message(FATAL_ERROR "Explicit ${required} is required")
  endif()
endforeach()
if(NOT OMNIPLAY_RUNTIME_ID MATCHES "^[A-Za-z0-9_-]+$" OR NOT OMNIPLAY_CORE_FILENAME MATCHES "^[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "Runtime identity and core filename must be literal basenames")
endif()
if(NOT OMNIPLAY_VIDEO_BACKEND MATCHES "^(software|opengl_core)$")
  message(FATAL_ERROR "A native worker must use an explicit supported video adapter")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE INTERNAL "")
set(JSON_Install OFF CACHE INTERNAL "")
if(NOT OMNIPLAY_DEVICE_FREE_ACTOR OR OMNIPLAY_BUILD_PRESENTATION_HOST)
  add_subdirectory("${OMNIPLAY_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
endif()
add_subdirectory("${OMNIPLAY_JSON_SOURCE}" "${CMAKE_BINARY_DIR}/json" EXCLUDE_FROM_ALL)
set(OMNIPLAY_PRESENTATION_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../presentation" CACHE PATH "Reviewed presentation wire/client source")
if(NOT EXISTS "${OMNIPLAY_PRESENTATION_SOURCE}/src/client.hpp")
  message(FATAL_ERROR "Explicit presentation client source is required")
endif()
add_subdirectory("${OMNIPLAY_PRESENTATION_SOURCE}" "${CMAKE_BINARY_DIR}/presentation" EXCLUDE_FROM_ALL)
configure_file(src/runtime_identity.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/runtime_identity.hpp" @ONLY)
add_executable(omniplay-native-worker src/main.cpp src/engine.cpp src/core_module.cpp src/protocol.cpp src/saved_files.cpp src/memory_files.cpp src/md_portable.cpp src/measurement.cpp)
target_include_directories(omniplay-native-worker PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated" "${OMNIPLAY_LIBRETRO_INCLUDE}")
target_compile_definitions(omniplay-native-worker PRIVATE SDL_MAIN_HANDLED RETRO_IMPORT_SYMBOLS NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(omniplay-native-worker PRIVATE nlohmann_json::nlohmann_json omniplay-presentation-client)
if(OMNIPLAY_GUEST_ACTOR)
  target_compile_definitions(omniplay-native-worker PRIVATE OMNIPLAY_GUEST_ACTOR=1)
  target_compile_definitions(omniplay-presentation-wire PRIVATE OMNIPLAY_GUEST_ACTOR=1)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(omniplay-native-worker PRIVATE ${CMAKE_DL_LIBS})
endif()
if(OMNIPLAY_DEVICE_FREE_ACTOR)
  target_compile_definitions(omniplay-native-worker PRIVATE OMNIPLAY_DEVICE_FREE_ACTOR=1)
  # SDL headers define the existing input wire's numeric scancodes/types only.
  # No SDL implementation or OpenGL object may enter this executable.
  target_include_directories(omniplay-native-worker PRIVATE "${OMNIPLAY_SDL_SOURCE}/include")
else()
  target_sources(omniplay-native-worker PRIVATE src/gl_video.cpp)
  target_link_libraries(omniplay-native-worker PRIVATE SDL3::SDL3-static)
endif()
if(MSVC)
  target_sources(omniplay-native-worker PRIVATE worker.manifest)
  target_compile_options(omniplay-native-worker PRIVATE /W4 /permissive- /GS /guard:cf /utf-8)
  target_link_options(omniplay-native-worker PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
else()
  target_compile_options(omniplay-native-worker PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_options(omniplay-native-worker PRIVATE -fPIE)
  target_link_options(omniplay-native-worker PRIVATE -pie -Wl,-z,relro,-z,now,-z,noexecstack)
  set_target_properties(omniplay-native-worker PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set_target_properties(omniplay-native-worker PROPERTIES OSX_ARCHITECTURES arm64 OSX_DEPLOYMENT_TARGET 11.0)
  target_compile_options(omniplay-native-worker PRIVATE -fstack-protector-strong)
  target_link_options(omniplay-native-worker PRIVATE -Wl,-pie -Wl,-dead_strip -Wl,-fatal_warnings)
  set_target_properties(omniplay-native-worker PROPERTIES
    BUILD_RPATH "" INSTALL_RPATH "" XCODE_ATTRIBUTE_ENABLE_HARDENED_RUNTIME YES
    XCODE_ATTRIBUTE_CODE_SIGN_INJECT_BASE_ENTITLEMENTS NO
    XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/macos-actor.entitlements")
endif()
# A root-level aggregate keeps multi-project Visual Studio builds and Ninja on
# the same worker/helper dependency graph. EXCLUDE_FROM_ALL helper projects do
# not otherwise expose a directly buildable project in the root VS solution.
add_custom_target(omniplay-native-runtime-build DEPENDS omniplay-native-worker)
if(OMNIPLAY_BUILD_PRESENTATION_HOST)
  add_dependencies(omniplay-native-runtime-build omniplay-presentation-host)
endif()
