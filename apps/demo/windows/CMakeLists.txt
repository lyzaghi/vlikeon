cmake_minimum_required(VERSION 3.25)
project(VlikeonDemo LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The playable demo requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(VLIKEON_WEBVIEW2_SDK "" CACHE PATH "Pinned Microsoft.Web.WebView2 1.0.3537.50 SDK")
if(NOT EXISTS "${VLIKEON_WEBVIEW2_SDK}/build/native/include/WebView2.h")
  message(FATAL_ERROR "Supply the exact reviewed WebView2 SDK directory")
endif()
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
configure_file(demo.rc.in "${CMAKE_CURRENT_BINARY_DIR}/demo.rc" @ONLY)
add_executable(VlikeonDemo WIN32 main.cpp demo.manifest "${CMAKE_CURRENT_BINARY_DIR}/demo.rc")
target_include_directories(VlikeonDemo PRIVATE "${VLIKEON_WEBVIEW2_SDK}/build/native/include")
target_compile_definitions(VlikeonDemo PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(VlikeonDemo PRIVATE /W4 /EHsc /utf-8 /GS /guard:cf)
target_link_options(VlikeonDemo PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
target_link_libraries(VlikeonDemo PRIVATE "${VLIKEON_WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib" ole32 oleaut32 user32 shell32 shlwapi version advapi32)
